#include<iostream>
#include<cstring>
#include<cstdlib>
using namespace std;
class str
{
	char *p;

	//////////////////////////////////////////////////////

	public:
	str()
	{
		p=new char[100];
	}

	//////////////////////////////////////////////////////

	str(const char *s)
	{
		p=new char[strlen(s)+1];	
		strcpy(p,s);
	}

	////////////////////////////////////////////////////////

	str( str &s)
	{
		p=new char[strlen(s.p)+1];
		strcpy(p,s.p);
	}

	/////////////////////////////////////////////////////////

	~str()
	{
		delete []p;
	}
	////////////////////assignment/////////////////////////////////// 
	str &operator=(const str &s)
	{
		delete[] p;
		p=new char [strlen(s.p)+1];
		strcpy(p,s.p);
		return *this;
	}
	////////////////////////[] operator//////////////////////////////
	char &operator[] (int i)
	{
		return p[i];
	}
	/////////////////////////+ operator//////////////////////////
	str operator+(const str &s)
	{
		str temp;
		delete[] temp.p;
		temp.p=new char[strlen(p)+strlen(s.p)+1];
		strcpy(temp.p,p);
		strcat(temp.p,s.p);
		return temp;
	}
	//////////////////////////////> >= < <= == !=////////////////////////
	bool operator>(const str &s)
	{
		return strcmp(p,s.p)>0;
	}

	bool operator<(const str &s)
	{
		return strcmp(p,s.p)<0;}
	bool operator >=(const str &s)
	{
		return strcmp(p,s.p)>=0;
	}
	bool operator<=(const str &s)
	{
		return strcmp(p,s.p)<=0;
	}

	bool operator==(const str &s)
	{
		return strcmp(p,s.p)==0;
	}
	bool operator!=(const str &s)
	{
		return strcmp(p,s.p)!=0;
	}

	//////////////////////////////////////////////////////////

	friend istream& operator >> (istream& in ,str& s);
	friend ostream& operator << (ostream& out ,str& s);
	friend void len(str &s);
	friend void rev(str &s);
	friend void sub(str &s);
	friend void upper(str &s);
	friend void lower(str &s);
	friend void my_strchr(str &s ,char q);
	friend void my_strrchr(str &s,char q);
	friend void my_strcpy(str &s,str &r);
	friend void my_strcmp(str &s,str &r);
	friend void my_strcat(str &s,str &r);
	friend void my_strncmp(str &s,str &r,int c);
	friend void my_strncpy(str &s,str &r,int c);
	friend void my_strncat(str &s,str &r,int c);


};

void len(str &s)
{
	int i,c=0;
	for(i=0;s[i]!='\0';i++)
		c++;
	cout<<"length is: "<<c<<endl;
	cout<<" "<<endl;
}

void rev(str &s)
{
	char t;
	int i,j,len=0;
	for(len=0;s[len]!='\0';len++);
	for(i=0,j=len-1;i<j;i++,j--)
	{
		t=s[i];
		s[i]=s[j];
		s[j]=t;
	}
	cout<<"rev:"<<s<<endl;
	cout<< "  "<<endl;
}

void my_strcmp(str &s ,str &r)
{
	int i;
	for(i=0;s[i]!='\0'&&r[i]!='\0';i++)
	{
		if(s[i]!=r[i])
			break;
	}
	if(s[i]=='\0'&&r[i]=='\0')
		cout<<"strings are equal"<<endl;
	else
		cout<<"strings are unequal"<<endl;
	cout<<" "<<endl;
}

void my_strcat(str &s, str &r)
{
	int i, j;
	for(i = 0; s[i] != '\0'; i++);
	for(j = 0; r[j] != '\0'; j++)
		s[i++] = r[j];
	s[i] = '\0';
	cout << "after concatenate: " << s << endl;
}
